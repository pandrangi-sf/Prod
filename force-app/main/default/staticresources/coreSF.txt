var guid = (function() {
    function s4() {
        return Math.floor((1 + Math.random()) * 0x10000).toString(16).substring(1);
    }

    return function() {
        return s4() + s4() + '-' + s4() + '-' + s4() + '-' + s4() + '-' + s4() + s4() + s4();
    };
})();

function addFieldErrors($scope,$validate_email) {
    sw = true;
    if(typeof $validate_email == 'undefined' )
        $validate_email = false;
    $(".form-controls").removeClass('error');
    $(".form-controls").each(function() {
        if ($(this).find('.required').val() == "") {
            $(this).addClass('error');
            sw = false;
        }else{
            if($validate_email)
                if(!validateEmail($scope.Email)){
                    $(".email").parent().addClass('error');
                    sw = false;
                }
        }
      
    });
    
    return sw;
}

function validateEmail(email) {
    var re = /^(([^<>()[\]\\.,;:\s@\"]+(\.[^<>()[\]\\.,;:\s@\"]+)*)|(\".+\"))@((\[[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\])|(([a-zA-Z\-0-9]+\.)+[a-zA-Z]{2,}))$/;
    return re.test(email);
}

function startController($scope, $filter, $routeParams, $location) {
    $scope.$parent.downloadguideText="forceHide";
    $scope.$parent.visible1 = 1; 
    $scope.$parent.active1 = 1; 
    $scope.$parent.visible2 = 0; 
    $scope.$parent.active2 = 0; 
    $scope.$parent.visible3 = 0; 
    $scope.$parent.active3 = 0; 
    $scope.$parent.img1 = '';
    $scope.$parent.img2 = 'hide';
    $scope.$parent.img3 = 'hide';
    $scope.next = function() {
        if (addFieldErrors($scope,true)) {
            sendData($scope.FirstName,$scope.LastName, $scope.Email, $scope.$parent.guid);
            $.cookie('firstName', $scope.FirstName);
            $.cookie('lastName', $scope.LastName);
            $.cookie('email', $scope.Email);
            $.cookie('guid', $scope.$parent.guid);
            $location.path('about');
        }
        return false;
    };
}
